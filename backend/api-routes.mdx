## Chat Routes

### POST /api/chat/

Main conversational endpoint that orchestrates the entire flow.

**File:** `app/api/routes/chat.py`

```python
from fastapi import APIRouter, HTTPException
from app.models.schemas import ChatRequest, ChatResponse
from app.llm_engine.classifier import get_classifier_instance
from app.llm_engine.gemini_client import GeminiClient
from app.db.mongo import db
from app.core.config import settings

router = APIRouter(prefix="/api/chat", tags=["Chat"])

# Initialize classifier and Gemini client
classifier = get_classifier_instance(settings)
gemini_client = GeminiClient(api_key=settings.GEMINI_API_KEY)

@router.post("/", response_model=ChatResponse)
async def chat_endpoint(request: ChatRequest) -> ChatResponse:
    """
    Main chat endpoint.
    
    Process:
    1. Classify user intent (dining/announcement/general)
    2. Fetch relevant context from database
    3. Generate natural language response with Gemini
    
    Args:
        request: ChatRequest with message and optional user_id
        
    Returns:
        ChatResponse with reply text and source intent
        
    Raises:
        HTTPException: If database connection fails or Gemini API error
    """
    try:
        # Step 1: Classify intent
        intent = await classifier.classify(request.message)
        logger.info(f"Classified intent: {intent} for message: {request.message[:50]}")
        
        # Step 2: Fetch context based on intent
        if intent == "dining":
            context_data = await _fetch_dining_context()
        elif intent == "announcement":
            context_data = await _fetch_announcements_context()
        else:
            context_data = "General conversation. No specific context needed."
        
        # Step 3: Generate response
        reply = await gemini_client.generate_response(
            system_instruction=SYSTEM_INSTRUCTION,
            user_query=request.message,
            context_data=context_data
        )
        
        return ChatResponse(reply=reply, source=intent)
        
    except Exception as e:
        logger.error(f"Chat endpoint error: {e}")
        raise HTTPException(
            status_code=500,
            detail=f"Internal server error: {str(e)}"
        )

async def _fetch_dining_context() -> str:
    """Fetch today's dining menu from database."""
    today = datetime.now().strftime("%Y-%m-%d")
    
    menu = await db.db["yemekhane_listesi"].find_one({"date": today})
    
    if not menu:
        return "No menu available for today."
    
    # Format menu as text
    context = f"""Today's Menu ({menu['date']}):
    
Soup: {menu.get('soup', 'N/A')}

Main Courses:
{chr(10).join(f"• {item}" for item in menu.get('main_courses', []))}

Sides:
{chr(10).join(f"• {item}" for item in menu.get('sides', []))}
"""
    return context

async def _fetch_announcements_context() -> str:
    """Fetch recent announcements from database."""
    cursor = (
        db.db["cse_akdeniz_announcements"]
        .find()
        .sort("created_at", -1)
        .limit(5)
    )
    
    announcements = await cursor.to_list(length=5)
    
    if not announcements:
        return "No recent announcements available."
    
    # Format announcements as text
    context = "Recent Announcements:\n\n"
    for i, ann in enumerate(announcements, 1):
        context += f"{i}. {ann['title']}\n"
        context += f"   {ann.get('content', 'No content')[:200]}...\n"
        context += f"   URL: {ann['url']}\n\n"
    
    return context
```

## Dining Routes

### GET /api/dining/latest

Get the most recent dining menu.

```python
from fastapi import APIRouter, HTTPException
from app.models.schemas import DiningMenuResponse
from app.db.mongo import db

router = APIRouter(prefix="/api/dining", tags=["Dining"])

@router.get("/latest", response_model=DiningMenuResponse)
async def get_latest_dining_menu():
    """
    Get the most recent dining menu.
    
    Returns:
        DiningMenuResponse with menu details
        
    Raises:
        HTTPException: If no menu found (404)
    """
    menu = await db.db["yemekhane_listesi"].find_one(
        sort=[("date", -1)]
    )
    
    if not menu:
        raise HTTPException(status_code=404, detail="No menu found")
    
    return DiningMenuResponse(**menu)
```

### GET /api/dining/{date}

Get dining menu for a specific date.

```python
@router.get("/{date_str}", response_model=DiningMenuResponse)
async def get_dining_menu_by_date(date_str: str):
    """
    Get dining menu for a specific date.
    
    Args:
        date_str: Date in YYYY-MM-DD format (e.g., "2024-05-25")
        
    Returns:
        DiningMenuResponse with menu details
        
    Raises:
        HTTPException: If date format invalid or no menu found
    """
    # Validate date format
    try:
        datetime.strptime(date_str, "%Y-%m-%d")
    except ValueError:
        raise HTTPException(
            status_code=400,
            detail="Invalid date format. Use YYYY-MM-DD"
        )
    
    menu = await db.db["yemekhane_listesi"].find_one({"date": date_str})
    
    if not menu:
        raise HTTPException(
            status_code=404,
            detail=f"No menu found for {date_str}"
        )
    
    return DiningMenuResponse(**menu)
```

### GET /api/dining/range

Get dining menus for a date range.

```python
from datetime import datetime, timedelta
from typing import List

@router.get("/range/", response_model=List[DiningMenuResponse])
async def get_dining_menu_range(
    start_date: str,
    end_date: str,
    limit: int = 10
):
    """
    Get dining menus for a date range.
    
    Args:
        start_date: Start date (YYYY-MM-DD)
        end_date: End date (YYYY-MM-DD)
        limit: Maximum number of menus to return (default: 10)
        
    Returns:
        List of DiningMenuResponse objects
    """
    cursor = (
        db.db["yemekhane_listesi"]
        .find({
            "date": {
                "$gte": start_date,
                "$lte": end_date
            }
        })
        .sort("date", 1)
        .limit(limit)
    )
    
    menus = await cursor.to_list(length=limit)
    
    return [DiningMenuResponse(**menu) for menu in menus]
```

## Announcement Routes

### GET /api/announcements/

Get latest announcements.

```python
from fastapi import APIRouter, Query
from typing import List, Optional
from app.models.schemas import AnnouncementResponse
from app.db.mongo import db

router = APIRouter(prefix="/api/announcements", tags=["Announcements"])

@router.get("/", response_model=List[AnnouncementResponse])
async def get_latest_announcements(
    limit: int = Query(10, ge=1, le=50, description="Number of announcements"),
    skip: int = Query(0, ge=0, description="Number to skip for pagination")
):
    """
    Get latest announcements with pagination.
    
    Args:
        limit: Number of announcements to return (1-50)
        skip: Number of announcements to skip
        
    Returns:
        List of AnnouncementResponse objects
    """
    cursor = (
        db.db["cse_akdeniz_announcements"]
        .find()
        .sort("created_at", -1)
        .skip(skip)
        .limit(limit)
    )
    
    announcements = await cursor.to_list(length=limit)
    
    return [AnnouncementResponse(**ann) for ann in announcements]
```

### GET /api/announcements/{id}

Get specific announcement by ID.

```python
from bson import ObjectId
from bson.errors import InvalidId

@router.get("/{announcement_id}", response_model=AnnouncementResponse)
async def get_announcement_by_id(announcement_id: str):
    """
    Get specific announcement by MongoDB ObjectId.
    
    Args:
        announcement_id: MongoDB ObjectId as string
        
    Returns:
        AnnouncementResponse
        
    Raises:
        HTTPException: If ID invalid or not found
    """
    try:
        object_id = ObjectId(announcement_id)
    except InvalidId:
        raise HTTPException(
            status_code=400,
            detail="Invalid announcement ID format"
        )
    
    announcement = await db.db["cse_akdeniz_announcements"].find_one(
        {"_id": object_id}
    )
    
    if not announcement:
        raise HTTPException(
            status_code=404,
            detail="Announcement not found"
        )
    
    return AnnouncementResponse(**announcement)
```

### GET /api/announcements/search/

Search announcements by keyword.

```python
@router.get("/search/", response_model=List[AnnouncementResponse])
async def search_announcements(
    q: str = Query(..., min_length=1, description="Search query"),
    limit: int = Query(10, ge=1, le=50)
):
    """
    Search announcements by keyword in title or content.
    
    Args:
        q: Search query
        limit: Maximum results to return
        
    Returns:
        List of matching AnnouncementResponse objects
    """
    # MongoDB text search
    cursor = (
        db.db["cse_akdeniz_announcements"]
        .find({
            "$or": [
                {"title": {"$regex": q, "$options": "i"}},
                {"content": {"$regex": q, "$options": "i"}}
            ]
        })
        .sort("created_at", -1)
        .limit(limit)
    )
    
    announcements = await cursor.to_list(length=limit)
    
    return [AnnouncementResponse(**ann) for ann in announcements]
```

## Health Check Route

### GET /health

Simple health check endpoint for monitoring.

```python
from fastapi import FastAPI
from datetime import datetime

app = FastAPI()

@app.get("/health")
async def health_check():
    """
    Health check endpoint for monitoring and load balancers.
    
    Returns:
        Health status and metadata
    """
    try:
        # Verify database connection
        await db.client.admin.command('ping')
        db_status = "connected"
    except Exception as e:
        db_status = f"error: {str(e)}"
    
    return {
        "status": "healthy" if db_status == "connected" else "unhealthy",
        "timestamp": datetime.now().isoformat(),
        "database": db_status,
        "version": "1.0.0",
        "uptime_seconds": time.time() - startup_time
    }
```

## Error Handling Middleware

### Global Exception Handler

```python
from fastapi import Request, status
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """
    Handle Pydantic validation errors with custom response.
    """
    return JSONResponse(
        status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
        content={
            "detail": exc.errors(),
            "message": "Request validation failed"
        }
    )

@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """
    Catch-all exception handler for unexpected errors.
    """
    logger.error(f"Unhandled exception: {exc}", exc_info=True)
    
    return JSONResponse(
        status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
        content={
            "detail": "Internal server error",
            "message": str(exc) if settings.DEBUG else "An error occurred"
        }
    )
```

## Request/Response Models

### Pydantic Schemas

Located in `app/models/schemas.py`:

```python
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime

class ChatRequest(BaseModel):
    """Request schema for chat endpoint."""
    message: str = Field(..., min_length=1, max_length=1000)
    user_id: Optional[str] = Field(None, max_length=100)

class ChatResponse(BaseModel):
    """Response schema for chat endpoint."""
    reply: str
    source: str  # "dining" | "announcement" | "general"

class DiningMenuResponse(BaseModel):
    """Response schema for dining menu."""
    date: str
    soup: str
    main_courses: List[str]
    sides: List[str]
    created_at: datetime

    class Config:
        json_schema_extra = {
            "example": {
                "date": "2024-05-25",
                "soup": "Mercimek Çorbası",
                "main_courses": ["Tavuk Şinitzel", "Pilav"],
                "sides": ["Salata", "Ayran"],
                "created_at": "2024-05-24T08:00:00Z"
            }
        }

class AnnouncementResponse(BaseModel):
    """Response schema for announcements."""
    title: str
    content: str
    url: str
    date: Optional[str]
    created_at: datetime

    class Config:
        json_schema_extra = {
            "example": {
                "title": "Final Sınav Tarihleri",
                "content": "Final sınavları 15 Haziran'da başlayacak...",
                "url": "https://cse.akdeniz.edu.tr/duyuru/123",
                "date": "2024-05-20",
                "created_at": "2024-05-20T14:30:00Z"
            }
        }
```
